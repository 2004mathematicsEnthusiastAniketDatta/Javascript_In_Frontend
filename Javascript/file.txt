// 1. File API (Browser)
// Reading files from input element
const fileInput = document.createElement('input');
fileInput.type = 'file';
fileInput.onchange = (event) => {
    const file = event.target.files[0];
    const reader = new FileReader();
    
    reader.onload = (e) => {
        const contents = e.target.result;
        console.log('File contents:', contents);
    };
    
    reader.readAsText(file); // For text files
    // reader.readAsDataURL(file); // For binary files as base64
    // reader.readAsArrayBuffer(file); // For binary data
};

// 2. Saving files in browser
function saveFile(content, filename, type) {
    const blob = new Blob([content], { type: type || 'text/plain' });
    const url = URL.createObjectURL(blob);
    
    const a = document.createElement('a');
    a.href = url;
    a.download = filename;
    a.click();
    
    URL.revokeObjectURL(url);
}
